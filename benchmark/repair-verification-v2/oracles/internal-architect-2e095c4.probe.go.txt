package architect

// Behavioural contract probe for RunWhy.
//
// Caller is an interface, so every observable is reachable: which tool is asked,
// with which arguments, and what comes back. The probe never names a private
// helper and never asserts the exact rendering -- only that what the graph holds
// about the rule is present in what the reader is shown.

import (
	"errors"
	"strings"
	"testing"

	"github.com/globulario/sensei-code/internal/sensei"
)

// whyCaller answers awareness_resolve for the classes it was told to know, and
// records every call so the arguments can be asserted.
type whyCaller struct {
	knows map[string]map[string]any // class -> node
	calls []map[string]any
}

func (c *whyCaller) CallTool(name string, args map[string]any) (sensei.ToolResult, error) {
	c.calls = append(c.calls, args)
	class, _ := args["class"].(string)
	node, ok := c.knows[class]
	if !ok {
		return sensei.ToolResult{}, errors.New("nothing under that class")
	}
	return sensei.ToolResult{Structured: map[string]any{"node": node}}, nil
}

func node(id, label string) map[string]any {
	return map[string]any{"id": id, "label": label, "severity": "critical", "status": "active"}
}

// An empty id is refused, and the refusal says what the caller should have
// typed.
func TestContractAnEmptyIDIsRefusedWithTheExpectedForm(t *testing.T) {
	for _, id := range []string{"", "   "} {
		out, err := RunWhy(&whyCaller{}, "repo", id)
		if err == nil {
			t.Fatalf("RunWhy(%q) returned %q with no error", id, out)
		}
		if !strings.Contains(err.Error(), "/why") {
			t.Errorf("the refusal does not show the expected form: %v", err)
		}
	}
}

// What the graph holds about the rule reaches the reader.
func TestContractAResolvedRuleIsReportedToTheReader(t *testing.T) {
	c := &whyCaller{knows: map[string]map[string]any{
		"invariant": node("sensei_code.publication.never_merges", "Sensei Code never merges its own work"),
	}}
	out, err := RunWhy(c, "repo", "sensei_code.publication.never_merges")
	if err != nil {
		t.Fatalf("RunWhy: %v", err)
	}
	for _, want := range []string{"sensei_code.publication.never_merges", "never merges its own work"} {
		if !strings.Contains(out, want) {
			t.Errorf("the report does not carry %q:\n%s", want, out)
		}
	}
}

// The class is not assumed from the id: a rule of another kind is still found.
//
// An architect copies whatever they were shown, so an id that looks like one
// kind and is another must still resolve rather than being refused on a guess.
func TestContractARuleOfAnotherKindIsStillFound(t *testing.T) {
	c := &whyCaller{knows: map[string]map[string]any{
		"failure_mode": node("sensei_code.some.failure", "a recorded way this broke"),
	}}
	out, err := RunWhy(c, "repo", "sensei_code.some.failure")
	if err != nil {
		t.Fatalf("a rule held under another class was not found: %v", err)
	}
	if !strings.Contains(out, "a recorded way this broke") {
		t.Errorf("the resolved node did not reach the reader:\n%s", out)
	}
}

// A prefixed id resolves to the same rule as the bare one.
//
// Architects copy ids in the form they were displayed, which sometimes carries
// the class as a prefix. The observable requirement is that both forms find the
// rule; how the prefix is removed is not the contract.
func TestContractAPrefixedIDResolvesToTheSameRule(t *testing.T) {
	for _, id := range []string{
		"sensei_code.publication.never_merges",
		"invariant:sensei_code.publication.never_merges",
	} {
		c := &whyCaller{knows: map[string]map[string]any{
			"invariant": node("sensei_code.publication.never_merges", "Sensei Code never merges its own work"),
		}}
		out, err := RunWhy(c, "repo", id)
		if err != nil {
			t.Fatalf("RunWhy(%q): %v", id, err)
		}
		if !strings.Contains(out, "never merges its own work") {
			t.Errorf("RunWhy(%q) did not resolve the rule:\n%s", id, out)
		}
		// And the id sent to Sensei carries no class prefix.
		if len(c.calls) == 0 {
			t.Fatal("awareness_resolve was never called")
		}
		if sent, _ := c.calls[0]["id"].(string); strings.Contains(sent, ":") {
			t.Errorf("the id sent to Sensei still carries a class prefix: %q", sent)
		}
	}
}

// The domain scopes the query when there is one, and is omitted when there is
// not.
func TestContractTheDomainScopesTheQueryOnlyWhenPresent(t *testing.T) {
	with := &whyCaller{knows: map[string]map[string]any{"invariant": node("x", "y")}}
	_, _ = RunWhy(with, "github.com/globulario/sensei-code", "x")
	if len(with.calls) == 0 {
		t.Fatal("awareness_resolve was never called")
	}
	if got := with.calls[0]["domain"]; got != "github.com/globulario/sensei-code" {
		t.Errorf("domain = %#v, want the one supplied", got)
	}

	without := &whyCaller{knows: map[string]map[string]any{"invariant": node("x", "y")}}
	_, _ = RunWhy(without, "", "x")
	if len(without.calls) == 0 {
		t.Fatal("awareness_resolve was never called")
	}
	if v, present := without.calls[0]["domain"]; present {
		t.Errorf("domain = %#v was sent when none was supplied", v)
	}
}

// Holding nothing under an id is reported as that, and names the id.
func TestContractAnUnknownIDIsReportedNotInvented(t *testing.T) {
	out, err := RunWhy(&whyCaller{knows: map[string]map[string]any{}}, "repo", "sensei_code.nothing.here")
	if err == nil {
		t.Fatalf("an unknown id produced a report rather than an error:\n%s", out)
	}
	if !strings.Contains(err.Error(), "sensei_code.nothing.here") {
		t.Errorf("the error does not name the id asked about: %v", err)
	}
}
